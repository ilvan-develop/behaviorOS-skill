## Implementation

### 1. Filesystem Helper (`lib/fs-helpers.js`)

Extract Windows-safe operations into a reusable module:

```js
import { existsSync, mkdirSync, renameSync, unlinkSync, writeFileSync, readFileSync } from 'fs';
import { dirname } from 'path';

/**
 * Atomic file replacement with Windows fallback.
 * Three-phase: unlink destination → rename temp → fallback to direct write.
 */
export function replaceFileAtomic(targetPath, content) {
  const tmp = `${targetPath}.tmp.${process.pid}.${Date.now()}`;
  writeFileSync(tmp, content, 'utf8');

  try {
    if (existsSync(targetPath)) unlinkSync(targetPath);
    renameSync(tmp, targetPath);
  } catch {
    // Windows fallback: non-atomic direct write
    if (existsSync(tmp)) unlinkSync(tmp);
    writeFileSync(targetPath, content, 'utf8');
  }
}

/**
 * Ensure directory exists before writing.
 */
export function ensureDir(dir) {
  if (!existsSync(dir)) mkdirSync(dir, { recursive: true });
}
```

**Rationale:** The atomic-replace logic already exists inline in `governance-state.js:170-179`. Extracting it avoids duplication and provides a single place to test Windows semantics.

### 2. Audit Rotation Engine (`lib/audit-rotation.js`)

```js
import { existsSync, readFileSync, renameSync, unlinkSync, statSync, appendFileSync, mkdirSync } from 'fs';
import { join } from 'path';
import { replaceFileAtomic, ensureDir } from './fs-helpers.js';

const DEFAULT_CONFIG = {
  maxFileSize: 10 * 1024 * 1024,  // 10MB
  maxEvents: 10000,                // 10K events
  archiveBase: 'archive',          // relative to audit dir
};

/**
 * Check if current.jsonl needs rotation.
 * Returns { rotate: boolean, reason: string }
 */
export function shouldRotate(auditDir, config = {}) {
  const { maxFileSize, maxEvents } = { ...DEFAULT_CONFIG, ...config };
  const currentPath = join(auditDir, 'current.jsonl');

  if (!existsSync(currentPath)) return { rotate: false };

  const stat = statSync(currentPath);
  if (stat.size >= maxFileSize) {
    return { rotate: true, reason: `size ${stat.size} >= ${maxFileSize}` };
  }

  // Count lines (events)
  const content = readFileSync(currentPath, 'utf8');
  const lines = content.split('\n').filter(l => l.trim());
  if (lines.length >= maxEvents) {
    return { rotate: true, reason: `events ${lines.length} >= ${maxEvents}` };
  }

  return { rotate: false };
}

/**
 * Rotate current.jsonl to archive.
 * Preserves all events, creates archive directory structure.
 */
export function rotate(auditDir, config = {}) {
  const { archiveBase } = { ...DEFAULT_CONFIG, ...config };
  const currentPath = join(auditDir, 'current.jsonl');

  if (!existsSync(currentPath)) return { rotated: false };

  const content = readFileSync(currentPath, 'utf8');
  if (!content.trim()) return { rotated: false };

  // Determine archive path: archive/YYYY/MM/audit-YYYY-MM-DD.jsonl
  const now = new Date();
  const YYYY = String(now.getFullYear());
  const MM = String(now.getMonth() + 1).padStart(2, '0');
  const DD = String(now.getDate()).padStart(2, '0');

  const archiveDir = join(auditDir, archiveBase, YYYY, MM);
  const archiveFile = join(archiveDir, `audit-${YYYY}-${MM}-${DD}.jsonl`);

  ensureDir(archiveDir);

  // If archive file exists, append to it (same day rotation)
  if (existsSync(archiveFile)) {
    appendFileSync(archiveFile, content, 'utf8');
  } else {
    replaceFileAtomic(archiveFile, content);
  }

  // Clear current.jsonl (write empty file, not delete)
  replaceFileAtomic(currentPath, '');

  return {
    rotated: true,
    archivePath: archiveFile,
    events: content.split('\n').filter(l => l.trim()).length,
  };
}
```

### 3. Integration with `oage-lib.js`

Add rotation check to `appendAudit()`:

```js
export function appendAudit(root, entry) {
  const auditPath = join(auditDir(root), 'current.jsonl');
  ensureDir(auditDir(root));

  // Check rotation before append
  const { rotate: needsRotation, reason } = shouldRotate(auditDir(root));
  if (needsRotation) {
    rotate(auditDir(root));
    // Record rotation event
    appendFileSync(auditPath, JSON.stringify({
      timestamp: new Date().toISOString(),
      event: 'audit_rotated',
      reason,
    }) + '\n');
  }

  const record = { timestamp: new Date().toISOString(), ...entry };
  appendFileSync(auditPath, JSON.stringify(record) + '\n');
  return record;
}
```
